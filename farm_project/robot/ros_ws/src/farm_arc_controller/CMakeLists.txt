cmake_minimum_required(VERSION 3.8)
project(farm_arc_controller)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(arc_controller SHARED src/arc_controller.cpp)
target_include_directories(arc_controller PRIVATE include)
ament_target_dependencies(arc_controller nav2_regulated_pure_pursuit_controller nav2_core pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
install(TARGETS arc_controller LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_arc_math test/test_arc_math.cpp)
  target_include_directories(test_arc_math PRIVATE include)
  add_test(NAME arc_math COMMAND test_arc_math)
  add_executable(test_arc_return test/test_arc_return.cpp)
  target_include_directories(test_arc_return PRIVATE include)
  add_test(NAME arc_return COMMAND test_arc_return)
endif()
ament_package()
