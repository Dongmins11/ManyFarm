cmake_minimum_required(VERSION 3.16)
project(farm_tag_amcl)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(nav2_amcl REQUIRED)
find_package(nav2_util REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(amcl src/amcl.cpp)
ament_target_dependencies(amcl nav2_amcl nav2_util rclcpp rclcpp_lifecycle std_msgs tf2_geometry_msgs)
target_link_libraries(amcl nlohmann_json::nlohmann_json Eigen3::Eigen)
target_compile_options(amcl PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS amcl DESTINATION lib/${PROJECT_NAME})
ament_package()
