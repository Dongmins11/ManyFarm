cmake_minimum_required(VERSION 3.8)
project(farm_wall_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(nav2_behaviors REQUIRED)
find_package(std_msgs REQUIRED)
add_library(wall_controller SHARED src/wall_plugins.cpp src/wait_plugins.cpp)
target_include_directories(wall_controller PRIVATE include)
ament_target_dependencies(wall_controller nav2_controller nav2_behaviors std_msgs nav2_regulated_pure_pursuit_controller nav2_core pluginlib tf2_geometry_msgs rclcpp_lifecycle geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
install(TARGETS wall_controller LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_station_docking test/test_station_docking.cpp)
 target_include_directories(test_station_docking PRIVATE include)
 add_test(NAME station_docking COMMAND test_station_docking)
 add_executable(test_wall_control test/test_wall_control.cpp)
 target_include_directories(test_wall_control PRIVATE include)
 add_test(NAME wall_control COMMAND test_wall_control)
endif()
ament_package()
