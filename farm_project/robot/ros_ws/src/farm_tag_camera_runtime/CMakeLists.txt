cmake_minimum_required(VERSION 3.16)
project(farm_tag_camera_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(class_loader REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core)
# 기존 컴포넌트를 사용하며 전용 프로세스의 종료 수명만 관리한다.
function(add_camera_runner name library component)
  add_executable(${name} src/safe_node_main.cpp)
  target_include_directories(${name} PRIVATE src)
  target_compile_definitions(${name} PRIVATE COMPONENT_LIBRARY="${library}" COMPONENT_CLASS="${component}")
  ament_target_dependencies(${name} rclcpp rclcpp_components class_loader)
  target_link_libraries(${name} TBB::tbb opencv_core)
  install(TARGETS ${name} DESTINATION lib/${PROJECT_NAME})
endfunction()
add_camera_runner(rectify_safe librectify.so "image_proc::RectifyNode")
add_camera_runner(apriltag_safe libAprilTagNode.so AprilTagNode)
ament_package()
